Routing — การกำหนดเส้นทาง

Routing คือการกำหนดว่า Server จะ ตอบสนองอย่างไร เมื่อผู้ใช้เข้าถึง URL และ HTTP Method ต่างๆ

HTTP Methods ที่ใช้บ่อย

GET app.get('/path', handler) ดึงข้อมูล
POST app.post('/path', handler) ส่งข้อมูล
PUT app.put('/path', handler) แก้ไขข้อมูล
DELETE app.delete('/path', handler) ลบข้อมูล
routes/users.js — ตัวอย่าง REST API
const express = require('express');
const app = express();
app.use(express.json()); // รับ JSON body

// 📥 GET — ดึงรายชื่อผู้ใช้ทั้งหมด
app.get('/users', (req, res) => {
  res.json([{ id: 1, name: 'พี่โจ' }]);
});

// 📥 GET — ดึงผู้ใช้ตาม id (Route Parameter)
app.get('/users/:id', (req, res) => {
  const { id } = req.params;
  res.json({ id, name: 'ผู้ใช้ #' + id });
});

// 📤 POST — สร้างผู้ใช้ใหม่
app.post('/users', (req, res) => {
  const { name } = req.body;
  res.status(201).json({ message: 'สร้างสำเร็จ', name });
});

// 🗑 DELETE — ลบผู้ใช้
app.delete('/users/:id', (req, res) => {
  res.json({ message: 'ลบผู้ใช้ #' + req.params.id + ' สำเร็จ' });
});
req.params รับค่าจาก URL เช่น /users/:id → req.params.id
req.query รับค่าจาก Query String เช่น ?page=2 → req.query.page
req.body รับข้อมูลจาก Body (ต้องใช้ express.json() ก่อน)

Middleware — ตัวกลางประมวลผล

Middleware คือฟังก์ชันที่ทำงาน ระหว่าง Request เข้ามาและ Response ออกไป สามารถใช้สำหรับ Logging, Authentication, Error Handling ฯลฯ

Flow การทำงาน

Request
Middleware 1
Middleware 2
Route Handler
Response
middleware.js — ตัวอย่างการใช้งาน
// ① Middleware ทั่วไป — ทุก Request จะผ่านตรงนี้
app.use((req, res, next) => {
  console.log(\`[\${new Date().toLocaleTimeString()}] \${req.method} \${req.url}\`);
  next(); // ส่งต่อไปยัง Middleware/Route ถัดไป
});

// ② Middleware สำหรับ Static Files
app.use(express.static('public'));

// ③ Middleware เฉพาะเส้นทาง
const authCheck = (req, res, next) => {
  if (!req.headers.authorization) {
    return res.status(401).json({ error: 'กรุณาเข้าสู่ระบบก่อน' });
  }
  next();
};
app.get('/dashboard', authCheck, (req, res) => {
  res.send('ยินดีต้อนรับสู่ Dashboard');
});

// ④ Error Handling Middleware (ต้องมี 4 parameters)
app.use((err, req, res, next) => {
  console.error(err.stack);
  res.status(500).json({ error: 'เกิดข้อผิดพลาดภายใน Server' });
});
ลำดับสำคัญมาก! Middleware และ Route ที่ประกาศ ก่อน จะทำงาน ก่อน เสมอ และอย่าลืมเรียก next() เพื่อส่งต่อ ไม่เช่นนั้น Request จะค้างอยู่ตลอดกาล

Response Methods — วิธีส่งข้อมูลกลับ

res.send() Text / HTML

ส่งข้อมูลแบบ Text หรือ HTML ตอบกลับไปยังผู้ใช้

res.send('<h1>Hello!</h1>')
res.json() JSON

ส่งข้อมูลแบบ JSON พร้อมตั้งค่า Content-Type อัตโนมัติ

res.json({ status: 'ok' })
res.status() HTTP Code

กำหนด HTTP Status Code ก่อนส่ง Response

res.status(404).json({ error: 'ไม่พบข้อมูล' })
res.redirect() Redirect

เปลี่ยนเส้นทางผู้ใช้ไปยัง URL อื่น

res.redirect('/home')
res.sendFile() File

ส่งไฟล์จาก Server ไปยังผู้ใช้โดยตรง

res.sendFile('index.html')
res.render() Template

เรนเดอร์ Template Engine เช่น EJS, Pug, Handlebars

res.render('index', { title: 'หน้าแรก' })

Express Router — แยกไฟล์ Route

เมื่อโปรเจกต์ใหญ่ขึ้น ควรแยก Route ออกเป็นไฟล์ย่อยๆ เพื่อให้โค้ด อ่านง่ายและดูแลง่าย

routes/users.js
const express = require('express');
const router = express.Router();

router.get('/', (req, res) => {
  res.json({ users: [] });
});

router.post('/', (req, res) => {
  res.status(201).json({ message: 'สร้างสำเร็จ' });
});

module.exports = router;
index.js — เชื่อม Router
const express = require('express');
const usersRouter = require('./routes/users');
const app = express();

app.use(express.json());

// ✅ เชื่อม Router — /users/... ทั้งหมด
app.use('/users', usersRouter);

app.listen(3000, () => {
  console.log('✅ Server started!');
});
